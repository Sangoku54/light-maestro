    // Selection overlay
    if (hasSelection) {
      const s1 = (Math.min(selectionStart!, selectionEnd!) - viewStart) / viewDuration * w;
      const s2 = (Math.max(selectionStart!, selectionEnd!) - viewStart) / viewDuration * w;
      ctx.fillStyle = 'rgba(168, 85, 247, 0.15)';
      ctx.fillRect(s1, waveTop, s2 - s1, waveH);
      ctx.strokeStyle = 'rgba(168, 85, 247, 0.5)';
      ctx.lineWidth = 1;
      ctx.setLineDash([4, 4]);
      ctx.strokeRect(s1, waveTop, s2 - s1, waveH);
      ctx.setLineDash([]);
      ctx.fillStyle = '#a855f7';
      ctx.font = '9px monospace';
      ctx.textAlign = 'left';
      const sTime = Math.min(selectionStart!, selectionEnd!);
      const eTime = Math.max(selectionStart!, selectionEnd!);
      const selDur = eTime - sTime;
      ctx.fillText(`${formatTimeSel(sTime)} -> ${formatTimeSel(eTime)} (${selDur.toFixed(1)}s)`, s1 + 4, waveTop + 12);
    }

    // Playhead
    if (currentTime >= viewStart - 0.1 && currentTime <= viewEnd + 0.1) {
      const px = ((currentTime - viewStart) / viewDuration) * w;
      ctx.beginPath();
      ctx.strokeStyle = '#ffffff';
      ctx.lineWidth = 2;
      ctx.moveTo(px, 0);
      ctx.lineTo(px, h);
      ctx.stroke();
      ctx.beginPath();
      ctx.fillStyle = '#ffffff';
      ctx.moveTo(px, rulerHeight);
      ctx.lineTo(px - 5, rulerHeight - 8);
      ctx.lineTo(px + 5, rulerHeight - 8);
      ctx.closePath();
      ctx.fill();
    }

  }, [waveform, beats, sections, duration, viewStart, viewEnd, viewDuration, zoomLevel, currentTime, hasSelection, selectionStart, selectionEnd, rulerHeight, waveTop, waveH, formatTimeSel]);

  // Draw minimap
  useEffect(() => {
    const canvas = minimapCanvasRef.current;
    const container = minimapContainerRef.current;
    if (!canvas || !container) return;
    const dpr = window.devicePixelRatio || 1;
    const rect = container.getBoundingClientRect();
    canvas.width = rect.width * dpr;
    canvas.height = rect.height * dpr;
    canvas.style.width = rect.width + 'px';
    canvas.style.height = rect.height + 'px';
    const ctx = canvas.getContext('2d');
    if (!ctx) return;
    ctx.scale(dpr, dpr);
    const w = rect.width;
    const h = rect.height;
    ctx.clearRect(0, 0, w, h);
    sections.forEach(section => {
      const x1 = (section.startTime / duration) * w;
      const x2 = (section.endTime / duration) * w;
      ctx.fillStyle = section.color + '40';
      ctx.fillRect(x1, 0, x2 - x1, h);
    });
    const step = Math.max(1, Math.floor(waveform.length / w));
    const midY = h / 2;
    ctx.beginPath();
    ctx.strokeStyle = '#a855f766';
    ctx.lineWidth = 0.6;
    for (let x = 0; x < w; x++) {
      const idx = Math.floor((x / w) * waveform.length);
      const amp = waveform[Math.min(idx, waveform.length - 1)] || 0;
      ctx.moveTo(x, midY - amp * h * 0.4);
      ctx.lineTo(x, midY + amp * h * 0.4);
    }
    ctx.stroke();
    const vpX1 = (viewStart / duration) * w;
    const vpX2 = (viewEnd / duration) * w;
    ctx.strokeStyle = '#a855f7';
    ctx.lineWidth = 1.5;
    ctx.strokeRect(vpX1, 0, vpX2 - vpX1, h);
    ctx.fillStyle = '#a855f720';
    ctx.fillRect(vpX1, 0, vpX2 - vpX1, h);
    if (currentTime > 0) {
      const px = (currentTime / duration) * w;
      ctx.beginPath();
      ctx.strokeStyle = '#ffffff';
      ctx.lineWidth = 1;
      ctx.moveTo(px, 0);
      ctx.lineTo(px, h);
      ctx.stroke();
    }
  }, [waveform, sections, duration, viewStart, viewEnd, currentTime]);

  const formatTime = (s: number) => {
    const m = Math.floor(s / 60);
    const sec = Math.floor(s % 60);
    return `${m}:${sec.toString().padStart(2, '0')}`;
  };

  const formatTimeSel = (s: number) => {
    const m = Math.floor(s / 60);
    const sec = (s % 60).toFixed(1);
    return `${m}:${sec.padStart(4, '0')}`;
  };

  const currentZoomPercent = Math.round((viewDuration / duration) * 100);
  const hasSelectionForRender = selectionStart !== null && selectionEnd !== null && Math.abs(selectionEnd - selectionStart) > 0.05;

  return (
    <div className="space-y-2">
      <div className="flex items-center gap-2 px-1">
        <button onClick={zoomOut} disabled={zoomLevel <= 1.05} className="p-1 rounded text-gray-400 hover:text-white hover:bg-gray-800 disabled:opacity-30 disabled:cursor-not-allowed" title="Zoom out">
          <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"><circle cx="11" cy="11" r="8"/><line x1="21" y1="21" x2="16.65" y2="16.65"/><line x1="8" y1="11" x2="14" y2="11"/></svg>
        </button>
        <button onClick={zoomToFit} disabled={zoomLevel <= 1.05} className="text-[10px] px-2 py-0.5 rounded text-gray-500 hover:text-gray-300 hover:bg-gray-800 disabled:opacity-30 disabled:cursor-not-allowed">Fit</button>
        <div className="flex-1 h-4 bg-gray-900/50 rounded-full overflow-hidden cursor-pointer relative" onClick={handleMinimapClick}>
          <div className="absolute inset-0 flex items-center px-0.5">
            <div className="h-2 bg-purple-600/30 rounded-full transition-all" style={{ width: `${currentZoomPercent}%` }} />
          </div>
        </div>
        <button onClick={zoomIn} disabled={zoomLevel >= 64} className="p-1 rounded text-gray-400 hover:text-white hover:bg-gray-800 disabled:opacity-30 disabled:cursor-not-allowed" title="Zoom in">
          <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2"><circle cx="11" cy="11" r="8"/><line x1="21" y1="21" x2="16.65" y2="16.65"/><line x1="11" y1="8" x2="11" y2="14"/><line x1="8" y1="11" x2="14" y2="11"/></svg>
        </button>
        <span className="text-[10px] text-gray-500 w-12 text-right font-mono">{zoomLevel >= 10 ? `${Math.round(zoomLevel)}x` : `${zoomLevel.toFixed(1)}x`}</span>
      </div>
      {zoomLevel > 1.05 && (
        <div ref={minimapContainerRef} className="w-full h-8 cursor-pointer rounded overflow-hidden" onClick={handleMinimapClick}>
          <canvas ref={minimapCanvasRef} className="w-full h-full" />
        </div>
      )}
      <div ref={mainContainerRef}
        className="w-full h-40 cursor-pointer relative rounded-lg overflow-hidden select-none border border-gray-800/50"
        onClick={handleClick}
        onMouseDown={handleMouseDown}
        onMouseMove={handleMouseMove}
        onMouseUp={handleMouseUp}
        onMouseLeave={handleMouseUp}
        onWheel={handleWheel}
        onContextMenu={handleContextMenu}
        style={{ cursor: isDragging ? 'grabbing' : isSelecting ? 'col-resize' : zoomLevel > 1.05 ? 'grab' : 'crosshair' }}>
        <canvas ref={mainCanvasRef} className="w-full h-full" />
      </div>
      {selContextMenu && hasSelectionForRender && (
        <div className="fixed z-50 bg-[#1a1a2e] border border-purple-700/50 rounded-lg shadow-xl py-1 min-w-[150px]" style={{ left: selContextMenu.x, top: selContextMenu.y }}>
          <div className="px-3 py-1.5 text-[10px] text-purple-400 uppercase tracking-wider border-b border-gray-800">
            Set section: {formatTimeSel(Math.min(selectionStart!, selectionEnd!))} -> {formatTimeSel(Math.max(selectionStart!, selectionEnd!))}
          </div>
          {(['intro', 'verse', 'chorus', 'bridge', 'drop', 'outro'] as const).map(type => (
            <button key={type} onClick={() => { onSelectionChange?.(selectionStart!, selectionEnd!, type); clearSelection(); }}
              className="w-full text-left px-3 py-1.5 text-xs text-gray-300 hover:bg-gray-800 capitalize flex items-center gap-2">
              <span className="w-2 h-2 rounded-full" style={{ backgroundColor: SECTION_COLORS[type] }} />{type}
            </button>
          ))}
        </div>
      )}
    </div>
  );
};
